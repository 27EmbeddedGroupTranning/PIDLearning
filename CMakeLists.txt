cmake_minimum_required(VERSION 3.16)
project(gimbal_pid_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(gimbal_sim src/main.cpp src/tuning.cpp)
target_include_directories(gimbal_sim PRIVATE include)
enable_testing()
add_executable(pid_tests tests/pid_tests.cpp)
target_include_directories(pid_tests PRIVATE include)
add_test(NAME pid_tests COMMAND pid_tests)
